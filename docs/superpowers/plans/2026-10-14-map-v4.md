## Что выяснилось при чтении кода (уточнения к спецификации)

- **Дуги уже наполовину есть.** В `drawEdges` вторичные связи рисуются через `arcPath` при наведении и выборе, длинные — отростками (`stub`). План сводится к фоновым дугам на рабочем масштабе; переключатель «Все связи» остаётся (в спецификации сказано «заменяют» — поправить в Задаче 13).
- **Стрелки сейчас двигают карту.** Когда стиль выбран, стрелки начнут ходить по графу; без выбора по-прежнему панорама. Это смена поведения.
- **Камера сайта неравномерна по осям.** `X = sc.x·k + tx`, а `Y = AXIS_H + ty + yRef·syOf(k)`: при `k = 1` по x масштаб в 5 раз больше обзорного, по y — в 1,25. Прототип считал зум равномерным. Поэтому в Задаче 8 линии строятся заново на каждый масштаб.
- **Исправленная ошибка прототипа.** Береговые линии не рисовались (функция `rdp` схлопывала контуры в точку). После исправления слой тяжелее: 101 тыс. точек, кадр при зуме 15–20 мс, p95 35–41 мс. Поэтому слой кэшируется в битмап (Задача 8).
- **Louvain-скрипт в сборку не входит.** Состав туманностей уже есть в прототипе, редактор правит YAML. Подсказку для нового стиля даёт `closestPart` по весам связей (Задача 6). Черновик Louvain остаётся в `docs/map-v4/prototype/scripts/`.
- **Лупа времени = краевые желоба.** Центр окна остаётся в рабочем масштабе камеры, а всё, что вылезло за края, сжато в узкие желоба по 56 px. Так «вся история остаётся на экране», и один жест колесом ведёт от «Всей истории» к эпохе. Это самая неоднозначная часть спецификации, поэтому в Задаче 11 стоит визуальная проверка «идти / не идти».
- **Шрифт.** В `index.html` подключён только `Spectral` курсив 400; названия туманностей — курсив 500 и 600.

```js
// js/trail.js
// Маршрут по карте — простой путь без петель. Источник правды — адрес (параметр trail),
// поэтому «Назад» и ссылки работают без отдельного хранилища.
export const TRAIL_MAX = 24;

export function pushStep(trail, id) {
  if (trail[trail.length - 1] === id) return trail;
  const at = trail.indexOf(id);
  const next = at >= 0 ? trail.slice(0, at + 1) : [...trail, id];
  return next.length > TRAIL_MAX ? next.slice(next.length - TRAIL_MAX) : next;
}

export const toParam = (trail) => trail.join(',');

export function parseTrail(param, has) {
  return String(param ?? '').split(',').map((s) => s.trim()).filter((id) => id && has(id)).reduce(pushStep, []);
}
```

```js
// js/graphnav.js
// Шаг по графу с клавиатуры. row(id) — строка стиля в древе: вертикальный порядок.
export function stepTarget(store, row, id, key) {
  const s = store.get(id);
  if (!s) return null;
  const known = (x) => !!x && !!store.get(x);
  const start = (x) => store.get(x).start;
  if (key === 'ArrowLeft') return known(s.parents[0]?.id) ? s.parents[0].id : null;
  if (key === 'ArrowRight') {
    const kids = s.children.map((c) => c.id).filter(known);
    const d = (x) => Math.abs(row(x) - row(id));
    return kids.sort((a, b) => d(a) - d(b) || start(a) - start(b) || a.localeCompare(b))[0] ?? null;
  }
  if (key === 'ArrowUp' || key === 'ArrowDown') {
    const pid = s.parents[0]?.id;
    const pool = known(pid)
      ? store.get(pid).children.map((c) => c.id).filter((x) => known(x) && store.get(x).parents[0]?.id === pid)
      : store.styles.filter((x) => !x.parents.length).map((x) => x.id);
    const sorted = [...new Set(pool)].sort((a, b) => row(a) - row(b) || start(a) - start(b) || a.localeCompare(b));
    return sorted[sorted.indexOf(id) + (key === 'ArrowDown' ? 1 : -1)] ?? null;
  }
  return null;
}
```

```js
// js/ambient.js
// Фоновые дуги на рабочем масштабе: вторичные связи, оба конца которых на экране. Не больше cap,
// самые короткие по времени первыми: они читаются как «рядом» и не превращаются в паутину.
export function pickAmbient(edges, { visible, span, cap = 40 }) {
  return edges
    .filter((e) => !e.primary && visible(e.par) && visible(e.ch))
    .sort((a, b) => span(a) - span(b))
    .slice(0, cap);
}
```

```js
// Состав туманностей: проверка, подсказка для нового стиля, цвета.
const HEX = /^#[0-9a-f]{6}$/i;
const W = { evolved: 1, influence: 0.7, reaction: 0.4, revival: 0.25 };

// Куда по связям ближе всего стиль, которого ещё нет в туманностях.
export function closestPart(id, styles, partOf) {
  const score = new Map();
  const add = (other, type) => { const p = partOf.get(other); if (p) score.set(p, (score.get(p) ?? 0) + (W[type] ?? 0.25)); };
  for (const s of styles) {
    if (s.id === id) for (const p of s.parents ?? []) add(p.id, p.type);
    else for (const p of s.parents ?? []) if (p.id === id) add(s.id, p.type);
  }
  return [...score].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))[0]?.[0] ?? null;
}

export function validateNebulae(doc, styles) {
  const errors = [], warnings = [];
  const ids = new Set(styles.map((s) => s.id));
  const nebIds = new Set(), partIds = new Set(), partOf = new Map(), place = new Map();
  for (const n of doc?.nebulae ?? []) {
    if (!n.id || !n.name) errors.push(`nebula without id or name: ${JSON.stringify(n.id ?? n.name)}`);
    if (nebIds.has(n.id)) errors.push(`duplicate nebula id "${n.id}"`);
    nebIds.add(n.id);
    if (n.color != null && !HEX.test(n.color)) errors.push(`nebula "${n.id}": color must be #rrggbb`);
    if (!n.parts?.length) errors.push(`nebula "${n.id}" has no parts`);
    for (const p of n.parts ?? []) {
      if (partIds.has(p.id)) errors.push(`duplicate part id "${p.id}"`);
      partIds.add(p.id);
      if (!p.styles?.length) errors.push(`part "${p.id}" has no styles`);
      for (const sid of p.styles ?? []) {
        if (!ids.has(sid)) { errors.push(`part "${p.id}": unknown style "${sid}"`); continue; }
        if (place.has(sid)) errors.push(`style "${sid}" is listed twice (${place.get(sid)} and part "${p.id}")`);
        place.set(sid, `part "${p.id}"`);
        partOf.set(sid, p.id);
      }
    }
  }
  for (const sid of doc?.free ?? []) {
    if (!ids.has(sid)) { errors.push(`free: unknown style "${sid}"`); continue; }
    if (place.has(sid)) errors.push(`style "${sid}" is listed twice (${place.get(sid)} and free)`);
    place.set(sid, 'free');
  }
  for (const s of styles) {
    if (place.has(s.id)) continue;
    const near = closestPart(s.id, styles, partOf);
    errors.push(`style "${s.id}" is not in any nebula part${near ? ` (closest by links: part "${near}")` : ''}; add it to content/nebulae.yaml or to free`);
  }
  return { errors, warnings };
```

```js
// js/lens.js
// Краевые «желоба»: в середине окна масштаб рабочий (x·k + tx), а всё, что вылезло за края,
// сжато в узкие желоба слева и справа — вся история остаётся на экране.
// Отображение непрерывно, монотонно и обратимо.
export function createGutterLens({ total, k, tx, width, gutter = 56, pad = 6 }) {
  const lin = (x) => x * k + tx;
  const hasL = tx < pad;
  const hasR = lin(total) > width - pad;
  const xL = hasL ? Math.min(total, (gutter - tx) / k) : 0;
  const xR = hasR ? Math.max(xL, (width - gutter - tx) / k) : total;
  const sL = hasL ? (gutter - pad) / Math.max(xL, 1e-6) : 0;
  const sR = hasR ? (gutter - pad) / Math.max(total - xR, 1e-6) : 0;
  const fwd = (x) => {
    if (hasL && x < xL) return pad + x * sL;
    if (hasR && x > xR) return width - gutter + (x - xR) * sR;
    return lin(x);
  };
  const inv = (px) => {
    if (hasL && px < gutter) return Math.max(0, (px - pad) / sL);
    if (hasR && px > width - gutter) return Math.min(total, xR + (px - (width - gutter)) / sR);
    return (px - tx) / k;
  };
  const zone = (x) => (hasL && x < xL ? 'L' : hasR && x > xR ? 'R' : 'C');
  return { fwd, inv, zone, hasL, hasR, xL, xR, gutter, width };
}
```

```js
    function flyTo(fromId, toId) {
      clearTimeout(flightTimer);
      const a = nodeById.get(fromId), b = nodeById.get(toId);
      if (!a || !b || reduceMotion()) { focusCamera(toId, 0); return; }

    }
```
